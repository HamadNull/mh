MFA Login with TOTP

import hmac, hashlib, struct, time, base64

USERNAME, PASSWORD = "student", "12345"
SECRET_KEY = "JBSWY3DPEHPK3PXP"

def totp():
    key = base64.b32decode(SECRET_KEY)
    msg = struct.pack(">Q", int(time.time()) // 30)
    h = hmac.new(key, msg, hashlib.sha1).digest()
    n = struct.unpack(">I", h[h[-1] & 15:][:4])[0] & 0x7FFFFFFF
    return str(n % 1000000).zfill(6)

print("=== MFA Login ===")
u, p = input("Username: "), input("Password: ")

if u == USERNAME and p == PASSWORD:
    print("Password: Correct")
    print("Test OTP:", totp())
    otp = input("Enter OTP: ")
    print("Login successful. Access granted." if hmac.compare_digest(otp, totp())
          else "Invalid OTP. Access denied.")
else:
    print("Invalid username or password. Access denied.")
الكود 2

Salted SHA-256 Password Hashing

import hashlib
import os

password = input("Enter password: ")

salt = os.urandom(16)

hashed = hashlib.sha256(salt + password.encode()).hexdigest()

print("\nSalt:", salt.hex())
print("Hashed password:", hashed)

login = input("\nEnter password again: ")

check = hashlib.sha256(salt + login.encode()).hexdigest()

if check == hashed:
    print("Login successful!")
else:
    print("Wrong password!")
